```solidity
// HealthRecord.sol
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract HealthRecord {
    struct Record {
        string ipfsHash;
        address owner;
        uint256 timestamp;
    }

    mapping(uint => Record) private records;
    mapping(uint => mapping(
        address => bool
    )) private access;

    event AccessGranted(uint recordId, address doctor);
    event RecordCreated(uint recordId, address owner);

    modifier onlyOwner(uint recordId) {
        require(
            records[recordId].owner == msg.sender,
            "Not record owner"
        );
        _;
    }

    function grantAccess(
        uint recordId,
        address doctor
    ) public onlyOwner(recordId) {
        access[recordId][doctor] = true;
        emit AccessGranted(recordId, doctor);
    }

    function getRecord(
        uint recordId
    ) public view returns (string memory) {
        require(
            access[recordId][msg.sender],
            "Access denied"
        );
        return records[recordId].ipfsHash;
    }
}
```
